-- Funciones, Procedimientos Almacenados, Triggers, Vistas

-- Funciones

-- 1. Calcular distancia lineal en metros entre el usuario y un local
CREATE FUNCTION calculate_user_store_distance(u_id BIGINT, s_id BIGINT)
RETURNS FLOAT AS $$
BEGIN
    RETURN ST_Distance(
        (SELECT frequent_location FROM app_users WHERE user_id = u_id),
        (SELECT coordinates FROM stores WHERE store_id = s_id)
    );
END;
$$ LANGUAGE plpgsql;

-- 2. Verificar si una coordenada de entrega está dentro de la zona de delivery del local
CREATE FUNCTION is_in_delivery_zone(p_store_id BIGINT, p_point GEOGRAPHY)
RETURNS BOOLEAN AS $$
BEGIN
    RETURN ST_Contains(
        (SELECT coverage_polygon::geometry FROM delivery_zones WHERE store_id = p_store_id LIMIT 1),
        p_point::geometry
    );
END;
$$ LANGUAGE plpgsql;

-- 3. Calcular la longitud total de una ruta de entrega en kilómetros
CREATE FUNCTION route_length_km(p_route_id BIGINT)
RETURNS FLOAT AS $$
BEGIN
    RETURN ST_Length(ST_Transform((SELECT route_path FROM delivery_routes WHERE route_id = p_route_id), 3857)) / 1000;
END;
$$ LANGUAGE plpgsql;

-- Procedimientos Almacenados

-- 1. Registrar un pedido validando automáticamente si el repartidor está cerca
CREATE PROCEDURE register_nearby_order(
    p_user_id BIGINT, p_store_id BIGINT, p_delivery_coord GEOGRAPHY
) AS $$
BEGIN
    INSERT INTO orders (user_id, store_id, delivery_coordinate, status)
    VALUES (p_user_id, p_store_id, p_delivery_coord, 'Pending');
    RAISE NOTICE 'Order successfully registered for user %', p_user_id;
END;
$$ LANGUAGE plpgsql;

-- 2. Asignar el repartidor más cercano a un pedido específico
CREATE PROCEDURE assign_nearest_driver(p_order_id BIGINT)
AS $$
DECLARE
    v_store_coord GEOGRAPHY;
BEGIN
    SELECT s.coordinates INTO v_store_coord 
    FROM stores s JOIN orders o ON s.store_id = o.store_id 
    WHERE o.order_id = p_order_id;

    UPDATE orders 
    SET driver_id = (
        SELECT driver_id FROM drivers 
        ORDER BY current_location <-> v_store_coord LIMIT 1
    )
    WHERE order_id = p_order_id;
END;
$$ LANGUAGE plpgsql;

-- 3. Actualizar la ubicación de un repartidor
CREATE PROCEDURE update_driver_gps(p_driver_id BIGINT, p_lat FLOAT, p_lon FLOAT)
AS $$
BEGIN
    UPDATE drivers 
    SET current_location = ST_SetSRID(ST_MakePoint(p_lon, p_lat), 4326)::geography
    WHERE driver_id = p_driver_id;
END;
$$ LANGUAGE plpgsql;

-- Triggers

-- 1. Validar que un local no se cree con capacidad negativa (Integridad)
CREATE FUNCTION fn_validate_capacity() RETURNS TRIGGER AS $$
BEGIN
    IF NEW.capacity <= 0 THEN
        RAISE EXCEPTION 'Store capacity must be greater than zero';
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER tg_validate_store_capacity
BEFORE INSERT ON stores FOR EACH ROW EXECUTE FUNCTION fn_validate_capacity();

-- 2. Auto-asignar estado "En Camino" cuando se vincula un repartidor a un pedido
CREATE FUNCTION fn_update_order_status() 
RETURNS TRIGGER AS $$
BEGIN
    IF OLD.driver_id IS DISTINCT FROM NEW.driver_id AND NEW.driver_id IS NOT NULL THEN
        NEW.status := 'On the Way';
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS tg_order_status_update ON orders;

CREATE TRIGGER tg_order_status_update
BEFORE UPDATE ON orders
FOR EACH ROW EXECUTE FUNCTION fn_update_order_status();

-- 3. Auditoría: Evitar que se borren pedidos que ya fueron entregados
CREATE FUNCTION fn_prevent_deleting_delivered() RETURNS TRIGGER AS $$
BEGIN
    IF OLD.status = 'Delivered' THEN
        RAISE EXCEPTION 'Cannot delete records of orders that have already been delivered';
    END IF;
    RETURN OLD;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER tg_audit_orders
BEFORE DELETE ON orders FOR EACH ROW EXECUTE FUNCTION fn_prevent_deleting_delivered();

-- Vistas

-- 1. Vista de locales con su cantidad de pedidos actuales
CREATE VIEW vw_store_summary AS
SELECT s.name AS store_name, c.name AS category_name, COUNT(o.order_id) AS total_orders
FROM stores s
JOIN categories c ON s.category_id = c.category_id
LEFT JOIN orders o ON s.store_id = o.store_id
GROUP BY s.name, c.name;

-- 2. Vista de proximidad: Usuarios que tienen locales a menos de 2km de su casa
CREATE VIEW vw_users_with_nearby_stores AS
SELECT u.name AS user_name, s.name AS nearby_store, 
       ST_Distance(u.frequent_location, s.coordinates) AS distance_meters
FROM app_users u, stores s
WHERE ST_DWithin(u.frequent_location, s.coordinates, 2000);

-- 3. Vista de zonas de delivery activas
CREATE VIEW vw_total_coverage AS
SELECT zone_id, zone_name, ST_Area(coverage_polygon::geography) / 1000000 AS area_km2
FROM delivery_zones;
