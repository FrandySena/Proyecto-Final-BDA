CREATE DATABASE tracefood_db;

CREATE EXTENSION IF NOT EXISTS postgis;

-- 1. CREACIÓN DE TABLAS 
-- ==========================================

CREATE TABLE categories (
    category_id BIGSERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    description TEXT
);

CREATE TABLE stores (
    store_id BIGSERIAL PRIMARY KEY,
    category_id BIGINT NOT NULL REFERENCES categories(category_id),
    name VARCHAR(150) NOT NULL,
    capacity INTEGER,
    coordinates GEOGRAPHY(Point, 4326) NOT NULL
);

CREATE TABLE delivery_zones (
    zone_id BIGSERIAL PRIMARY KEY,
    store_id BIGINT NOT NULL REFERENCES stores(store_id),
    zone_name VARCHAR(150) NOT NULL,
    coverage_polygon GEOMETRY(Polygon, 4326) NOT NULL
);

CREATE TABLE app_users (
    user_id BIGSERIAL PRIMARY KEY,
    name VARCHAR(150) NOT NULL,
    email VARCHAR(150) UNIQUE NOT NULL,
    phone VARCHAR(20),
    frequent_location GEOGRAPHY(Point, 4326)
);

CREATE TABLE drivers (
    driver_id BIGSERIAL PRIMARY KEY,
    name VARCHAR(150) NOT NULL,
    phone VARCHAR(20),
    current_location GEOGRAPHY(Point, 4326)
);

CREATE TABLE orders (
    order_id BIGSERIAL PRIMARY KEY,
    user_id BIGINT NOT NULL REFERENCES app_users(user_id),
    store_id BIGINT NOT NULL REFERENCES stores(store_id),
    driver_id BIGINT REFERENCES drivers(driver_id),
    delivery_coordinate GEOGRAPHY(Point, 4326) NOT NULL,
    order_date TIMESTAMPTZ DEFAULT NOW(),
    status VARCHAR(50) DEFAULT 'Pending'
);

CREATE TABLE delivery_routes (
    route_id BIGSERIAL PRIMARY KEY,
    order_id BIGINT NOT NULL REFERENCES orders(order_id),
    route_path GEOMETRY(LineString, 4326) NOT NULL
);
